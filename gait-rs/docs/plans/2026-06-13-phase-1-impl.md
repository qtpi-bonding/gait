# Phase 1 — Output Port Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give `gait run` a single proto-native stdout contract — the live event/delta stream and errors go to stderr, the final `TurnResult` envelope (`{message, result, changes}`) goes to stdout once — and extract `cmd_run` into a `cli/` module, behind a clean `TurnSink` output port.

**Architecture:** `Emitter`→`Recorder` (record = commit + publish). A read-time `classify(&StreamItem)` splits channel items into `Trace`/`Error`; a `TurnSink` adapter (`StdioSink`) routes Trace+Error→stderr and prints the proto result envelope→stdout. The result envelope gains a ground-truth `changes` field computed from a new `CheckpointRepo::diff`. Everything on stdout/stderr is a proto message serialized via pbjson — no ad-hoc `serde_json::json!`.

**Tech Stack:** Rust, tokio, prost + pbjson (proto JSON), git2, jsonschema. Tests via `cargo test`.

**Spec:** [`../specs/2026-06-13-stream-output-sink.md`](../specs/2026-06-13-stream-output-sink.md). Parent plan: [`2026-06-13-turn-io-architecture.md`](./2026-06-13-turn-io-architecture.md).

```rust
fn diff(&self, gait_dir: &Path, session_id: &str, sha: &str) -> anyhow::Result<Option<crate::proto::gait::log::v1::Changes>> {
    use crate::proto::gait::log::v1::Changes;
    let git_dir = checkpoint_git_dir(gait_dir, session_id);
    if !git_dir.join("HEAD").exists() { return Ok(None); }
    let shadow = Repository::open_bare(&git_dir)?;
    let commit = shadow.find_commit(Oid::from_str(sha)?)?;
    let new_tree = commit.tree()?;
    let parent_tree = match commit.parent(0) { Ok(p) => Some(p.tree()?), Err(_) => None };
    let diff = shadow.diff_tree_to_tree(parent_tree.as_ref(), Some(&new_tree), None)?;
    let (mut added, mut modified, mut deleted) = (vec![], vec![], vec![]);
    diff.foreach(&mut |delta, _| {
        let path = delta.new_file().path().or_else(|| delta.old_file().path())
            .map(|p| p.to_string_lossy().into_owned());
        if let Some(p) = path {
            match delta.status() {
                git2::Delta::Added | git2::Delta::Copied => added.push(p),
                git2::Delta::Deleted => deleted.push(p),
                _ => modified.push(p), // Modified, Renamed, Typechange
            }
        }
        true
    }, None, None, None)?;
    Ok(Some(Changes { added, modified, deleted }))
}
```

```rust
fn init(&self, gait_dir: &Path, session_id: &str, codebase: &Path) -> anyhow::Result<()> {
    let cb = Repository::open(codebase).ok(); // None for non-git — shadow still created, stores own objects

    let git_dir = checkpoint_git_dir(gait_dir, session_id);
    if git_dir.join("HEAD").exists() { return Ok(()); }
    std::fs::create_dir_all(&git_dir)?;
    let shadow = Repository::init_bare(&git_dir)?;

    // Borrow the codebase's object store ONLY when it is a git repo (blob-sharing
    // optimization). Non-git: the shadow keeps its own objects via add_all.
    if let Some(ref cb) = cb {
        let git_path = cb.path();
        let cb_objects_raw = {
            let commondir_file = git_path.join("commondir");
            if commondir_file.exists() {
                let rel = std::fs::read_to_string(&commondir_file).unwrap_or_default();
                git_path.join(rel.trim()).join("objects")
            } else { git_path.join("objects") }
        };
        let cb_objects = std::fs::canonicalize(&cb_objects_raw).unwrap_or(cb_objects_raw);
        let info_dir = git_dir.join("objects").join("info");
        std::fs::create_dir_all(&info_dir)?;
        std::fs::write(info_dir.join("alternates"), format!("{}\n", cb_objects.display()))?;
    }

    let exclude_dir = git_dir.join("info");
    std::fs::create_dir_all(&exclude_dir)?;
    std::fs::write(exclude_dir.join("exclude"), ".git\n")?;

    // Seed HEAD from the codebase's current commit when available; else leave
    // unborn so the first checkpoint is the root commit.
    if let Some(ref cb) = cb {
        if let Ok(head) = cb.head() {
            if let Ok(commit) = head.peel_to_commit() {
                shadow.reference("refs/heads/rewind", commit.id(), true, "init checkpoint")?;
                shadow.set_head("refs/heads/rewind")?;
            }
        }
    }
    Ok(())
}
```

```rust
//! Agnostic turn-output port: classify channel items + the TurnSink contract.
use crate::proto::gait::log::v1::{stream_item, payload, Event, StreamItem};
use std::io::Write;

#[derive(Debug, PartialEq, Eq, Clone, Copy)]
pub enum StreamClass { Trace, Error }

pub fn classify(item: &StreamItem) -> StreamClass {
    match &item.item {
        Some(stream_item::Item::Delta(_)) => StreamClass::Trace,
        Some(stream_item::Item::Event(ev)) => classify_event(ev),
        None => StreamClass::Trace,
    }
}

fn classify_event(ev: &Event) -> StreamClass {
    match ev.body.as_ref() {
        Some(crate::proto::gait::log::v1::event::Body::Payload(p)) => match &p.kind {
            Some(payload::Kind::Error(_)) => StreamClass::Error,
            _ => StreamClass::Trace,
        },
        _ => StreamClass::Trace,
    }
}

#[derive(Debug, PartialEq, Eq, Clone, Copy)]
pub enum Diag { Info, Error }

pub trait TurnSink {
    fn item(&mut self, item: &StreamItem);
    fn result(&mut self, r: &crate::turn::TurnResult);

}

/// Drive a StreamItem channel into a sink until it closes.
pub async fn drive(mut rx: tokio::sync::mpsc::UnboundedReceiver<StreamItem>, sink: &mut impl TurnSink) {
    while let Some(item) = rx.recv().await { sink.item(&item); }
}
```

```rust
//! CLI adapter for the output port: routes by class to stdout/stderr.
use std::io::Write;
use crate::port::output::{classify, Diag, StreamClass, TurnSink};
use crate::proto::gait::log::v1::StreamItem;
use crate::turn::TurnResult;

pub struct StdioSink<O: Write, E: Write> {
    out: O,
    err: E,
    quiet: bool,
    pub failed: bool,
}

impl StdioSink<std::io::Stdout, std::io::Stderr> {
    pub fn stdio(quiet: bool) -> Self {
        Self { out: std::io::stdout(), err: std::io::stderr(), quiet, failed: false }
    }
}

impl<O: Write, E: Write> StdioSink<O, E> {
    pub fn new(out: O, err: E, quiet: bool) -> Self { Self { out, err, quiet, failed: false } }
}

impl<O: Write, E: Write> TurnSink for StdioSink<O, E> {
    fn item(&mut self, item: &StreamItem) {
        match classify(item) {
            StreamClass::Error => {
                self.failed = true;
                if let Ok(j) = serde_json::to_string(item) { let _ = writeln!(self.err, "{j}"); }
            }
            StreamClass::Trace => {
                if !self.quiet {
                    if let Ok(j) = serde_json::to_string(item) { let _ = writeln!(self.err, "{j}"); }
                }
            }
        }
    }
    fn result(&mut self, r: &TurnResult) {
        // Phase-1 placeholder: Task 9 replaces this with proto RunResult serialization.
        let line = crate::cli::run::run_result_json(r);
        let _ = writeln!(self.out, "{line}");
    }
    fn session_id(&mut self, id: &str) {
        // Task 9 replaces with a proto message; minimal here for the test.
        let _ = writeln!(self.out, "{}", serde_json::json!({"session_id": id}));
    }
    fn diagnostic(&mut self, level: Diag, msg: &str) {
        if level == Diag::Error { self.failed = true; }
        if level == Diag::Error || !self.quiet { let _ = writeln!(self.err, "[gait] {msg}"); }
    }
}
```

```rust
#[derive(Default, Debug)]
pub struct RunArgs {
    pub session_id: Option<String>,
    pub task: Option<String>,
    pub auto: bool,
    pub model_override: Option<String>,
    pub agent_name: Option<String>,
    pub result_schema_path: Option<String>,
    pub allow: Vec<String>,
    pub deny: Vec<String>,
    pub max_iter: Option<usize>,
}

impl RunArgs {
    pub fn parse(argv: &[String]) -> Self {
        let mut a = RunArgs::default();
        let mut i = 0;
        while i < argv.len() {
            match argv[i].as_str() {
                "--session"       => { i += 1; a.session_id = argv.get(i).cloned(); }
                "--task"          => { i += 1; a.task = argv.get(i).cloned(); }
                "--model"         => { i += 1; a.model_override = argv.get(i).cloned(); }
                "--auto"          => { a.auto = true; }
                "--agent"         => { i += 1; a.agent_name = argv.get(i).cloned(); }
                "--result-schema" => { i += 1; a.result_schema_path = argv.get(i).cloned(); }
                "--allow"         => { i += 1; if let Some(p) = argv.get(i) { a.allow.push(p.clone()); } }
                "--deny"          => { i += 1; if let Some(p) = argv.get(i) { a.deny.push(p.clone()); } }
                "--max-iter"      => { i += 1; a.max_iter = argv.get(i).and_then(|s| s.parse().ok()); }
                "--gait-dir" | "--codebase" => { i += 1; } // consumed elsewhere
                _ => {}
            }
            i += 1;
        }
        a
    }
}
```
